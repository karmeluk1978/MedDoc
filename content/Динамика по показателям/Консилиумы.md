# Консилиумы — все сессии по датам

← [[Индекс_анализов]]

```dataviewjs
dv.container.classList.add("dyn-text");
const v = x => (x == null || x.length === 0) ? "—" : x;
const pages = dv.pages('#консилиум').sort(p => p.date, 'asc');
dv.table(
  ["Дата", "Учреждение", "Состав врачей", "Решение по ЖКТ", "Суть"],
  pages.map(p => [
    dv.fileLink(p.file.path, false, p.date.toFormat("dd.MM.yyyy")),
    v(p.institution),
    v(p.consilium_doctors),
    v(p.gi_decision),
    v(p.summary)
  ])
);
```
