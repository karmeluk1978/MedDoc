# Приём препаратов — вся динамика

← [[Индекс_анализов]]

```dataviewjs
dv.container.classList.add("vertical-headers");
const vh = t => `<span class="vh">${t}</span>`;
const v = x => (x == null || x.length === 0) ? "—" : x;
const pages = dv.pages('#приём_препаратов')
  .where(p => /^\d{4}-\d{2}-\d{2}$/.test(p.file.name))
  .sort(p => p.file.name, 'asc');
const names = [];
for (const p of pages) {
  for (const t of p.file.tasks) {
    const n = t.text.split(" — ")[0];
    if (!names.includes(n)) names.push(n);
  }
}
const rows = [];
for (const p of pages) {
  const m = {};
  let done = 0, total = 0;
  for (const t of p.file.tasks) {
    m[t.text.split(" — ")[0]] = t.completed ? "✅" : null;
    total++;
    if (t.completed) done++;
  }
  const txt = await dv.io.load(p.file.path);
  const mm = txt ? txt.match(/## Заметки[ \t]*\n([\s\S]*)$/) : null;
  const note = mm ? mm[1].trim().replace(/\n+/g, "<br>") : "";
  const d = p.date?.toFormat ? p.date.toFormat("dd.MM.yyyy") : p.file.name;
  rows.push([
    dv.fileLink(p.file.path, false, d),
    ...names.map(n => v(m[n])),
    `${done}/${total}`,
    v(note)
  ]);
}
dv.table(["Дата", ...names.map(n => vh(n)), "Принято", "Примечания"], rows);
```
