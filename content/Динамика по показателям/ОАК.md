# ОАК — вся динамика (ключевые показатели)

← [[Индекс_анализов]]

```dataviewjs
dv.container.classList.add("vertical-headers", "dysb-main");
const vh = t => `<span class="vh">${t}</span>`;
const v = x => (x == null || x.length === 0) ? "—" : x;
const cols = [["Лейкоциты", "лейкоциты"], ["Hb", "гемоглобин"], ["Тромбоциты", "тромбоциты"]];
const pages = dv.pages('#оак').sort(p => p.date, 'asc');
dv.table(
  ["Дата", "Учреждение", ...cols.map(c => vh(c[0])), "Отклонения"],
  pages.map(p => [
    dv.fileLink(p.file.path, false, p.date.toFormat("dd.MM.yyyy")),
    v(p.institution),
    ...cols.map(c => v(p.values?.[c[1]])),
    v(p.abnormal)
  ])
);
```
