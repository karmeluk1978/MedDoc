# Скрытая кровь — вся динамика

← [[Индекс_анализов]]

```dataviewjs
dv.container.classList.add("vertical-headers");
const vh = t => `<span class="vh">${t}</span>`;
const v = x => (x == null || x.length === 0) ? "—" : x;
const cols = [["Результат", "скрытая_кровь"]];
const pages = dv.pages('#скрытая_кровь').sort(p => p.date, 'asc');
dv.table(
  ["Дата", "Учреждение", ...cols.map(c => vh(c[0]))],
  pages.map(p => [
    dv.fileLink(p.file.path, false, p.date.toFormat("dd.MM.yyyy")),
    v(p.institution),
    ...cols.map(c => v(p.values?.[c[1]]))
  ])
);
```
