# Нейтральный жир / жирные кислоты — вся динамика (копрограмма)

← [[Индекс_анализов]]

```dataviewjs
dv.container.classList.add("vertical-headers");
const vh = t => `<span class="vh">${t}</span>`;
const norm = x => {
  if (x == null) return "—";
  const s = String(x).trim();
  if (!s) return "—";
  if (/^(отсутств|отрицат|не обнаруж)/i.test(s) || s === "-") return "нет";
  return s;
};
const show = x => { const s = norm(x); return s === "—" ? s : `<span>${s}</span>`; };
const pages = dv.pages('#копрограмма').sort(p => p.date, 'asc');
dv.table(
  ["Дата", "Учреждение", vh("Нейтр. жир"), vh("Жирные кислоты")],
  pages.map(p => [
    dv.fileLink(p.file.path, false, p.date.toFormat("dd.MM.yyyy")),
    p.institution ?? "—",
    show(p.values?.["нейтральный_жир"]),
    show(p.values?.["жирные_кислоты"])
  ])
);
```

> Значения берутся из поля `values` заметок копрограммы. «нет» — показатель определялся и оказался отрицательным («отсутствует», «отрицательный», «не обнаружено», «-» в источнике); «—» — этого показателя в анализе нет (не исследовался). «ед.» — единично.
