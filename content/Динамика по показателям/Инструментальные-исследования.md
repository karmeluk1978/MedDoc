# Инструментальные исследования — вся хронология (ЭКГ, УЗИ, рентген, КТ)

← [[Индекс_анализов]]

```dataviewjs
dv.container.classList.add("dyn-text");
const v = x => (x == null || x.length === 0) ? "—" : x;
const fmt = d => (d && d.toFormat) ? d.toFormat("dd.MM.yyyy") : "—";
const pages = dv.pages('"Хранилище/Медицинские документы"')
  .where(p => p.type == "инструментальное")
  .sort(p => (p.date && p.date.toMillis) ? p.date.toMillis() : Number.MAX_SAFE_INTEGER, 'asc');
dv.table(
  ["Дата", "Учреждение", "Категория", "Отклонения"],
  pages.map(p => [
    dv.fileLink(p.file.path, false, fmt(p.date)),
    v(p.institution),
    v(p.category),
    v(p.abnormal)
  ])
);
```
